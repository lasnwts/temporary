package ru.usb.pub_diasoft_cft_import.config;

import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.boot.autoconfigure.jdbc.DataSourceProperties;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.datasource.DataSourceTransactionManager;
import org.springframework.transaction.PlatformTransactionManager;

import javax.sql.DataSource;

@Configuration
public class DiasDataSourceConfig {

    /**
     * Читает базовые свойства spring.dias.datasource.* (url, username, password, driver-class-name).
     */
    @Bean
    @Primary
    @ConfigurationProperties("spring.dias.datasource")
    public DataSourceProperties diasDataSourceProperties() {
        return new DataSourceProperties();
    }

    /**
     * Отдельный бин HikariConfig, куда Spring забиндит только hikari-специфичные
     * свойства из spring.dias.datasource.hikari.* (maximum-pool-size,
     * connection-timeout, pool-name и т.д.).
     * <p>
     * Важно: сюда НЕ попадают url/username/password — их мы установим вручную
     * ниже, чтобы явно передать jdbcUrl.
     */
    @Bean
    @Primary
    @ConfigurationProperties("spring.dias.datasource.hikari")
    public HikariConfig diasHikariConfig() {
        return new HikariConfig();
    }

    /**
     * Собираем HikariDataSource из двух источников:
     * <ul>
     *   <li>{@link DataSourceProperties} — url/username/password/driver-class-name;</li>
     *   <li>{@link HikariConfig} — параметры пула из application.properties.</li>
     * </ul>
     * Явный {@code setJdbcUrl(props.getUrl())} устраняет проблему
     * "Driver ... claims to not accept jdbcUrl", которая возникала
     * при ручной конфигурации DataSource без этого вызова.
     */
    @Bean
    @Primary
    public HikariDataSource diasDataSource(
            @Qualifier("diasDataSourceProperties") DataSourceProperties props,
            @Qualifier("diasHikariConfig") HikariConfig hikariConfig) {

        // Явно проставляем подключение — обязательно ДО new HikariDataSource(cfg)
        hikariConfig.setJdbcUrl(props.getUrl());
        hikariConfig.setUsername(props.getUsername());
        hikariConfig.setPassword(props.getPassword());

        if (props.getDriverClassName() != null && !props.getDriverClassName().isBlank()) {
            hikariConfig.setDriverClassName(props.getDriverClassName());
        }

        return new HikariDataSource(hikariConfig);
    }

    /**
     * JdbcTemplate, используемый в NikoilCftOperationDao и StatusUpdater.
     */
    @Bean
    @Primary
    public JdbcTemplate jdbcTemplate(
            @Qualifier("diasDataSource") DataSource dataSource) {
        return new JdbcTemplate(dataSource);
    }

    /**
     * Менеджер транзакций — нужен для @Transactional в StatusUpdater.markSent().
     */
    @Bean
    @Primary
    public PlatformTransactionManager transactionManager(
            @Qualifier("diasDataSource") DataSource dataSource) {
        return new DataSourceTransactionManager(dataSource);
    }
}
